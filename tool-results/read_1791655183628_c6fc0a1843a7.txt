   256→// HOSTEL MODULE
   257→// ═══════════════════════════════════════════════════════════════
   258→export function HostelModule() {
   259→  const [rooms, setRooms] = useState<HostelRoom[]>(HOSTEL_ROOMS);
   260→  const [allocOpen, setAllocOpen] = useState(false);
   261→  const [allocForm, setAllocForm] = useState({ roomId: "", studentId: "", studentName: "" });
   262→  const { toast } = useToast();
   263→  const totalCapacity = rooms.reduce((s, r) => s + r.capacity, 0);
   264→  const totalOccupied = rooms.reduce((s, r) => s + r.occupied, 0);
   265→  const availableBeds = totalCapacity - totalOccupied;
   266→
   267→  const handleAllocate = () => {
   268→    if (!allocForm.roomId || !allocForm.studentId.trim()) {
   269→      toast({ title: "Missing info", description: "Please select a room and enter Student ID.", variant: "destructive" });
   270→      return;
   271→    }
   272→    const room = rooms.find(r => r.id === allocForm.roomId);
   273→    if (!room || room.occupied >= room.capacity) {
   274→      toast({ title: "Room full", description: "This room has no available beds.", variant: "destructive" });
   275→      return;
   276→    }
   277→    setRooms(rooms.map(r => r.id === allocForm.roomId ? { ...r, occupied: r.occupied + 1 } : r));
   278→    toast({ title: "Room Allocated ✓", description: `${allocForm.studentName || allocForm.studentId} assigned to ${room.roomNo}.` });
   279→    setAllocOpen(false);
   280→    setAllocForm({ roomId: "", studentId: "", studentName: "" });
   281→  };
   282→
   283→  return (
   284→    <div className="space-y-3 fade-in">
   285→      <div className="flex justify-between items-center flex-wrap gap-3">
   286→        <p className="text-sm text-muted-foreground">Manage hostel rooms, bed allocation, and mess.</p>
   287→        <Button className="bg-primary hover:bg-primary/90" onClick={() => setAllocOpen(true)}><Plus className="w-4 h-4 mr-1.5" />Allocate Room</Button>
   288→      </div>
   289→
   290→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   291→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-primary/10 p-1.5"><Home className="w-4 h-4 text-primary" /></div><div><p className="text-xs text-muted-foreground">Total Rooms</p><p className="text-xl font-bold">{rooms.length}</p></div></CardContent></Card>
