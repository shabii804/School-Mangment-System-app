    31→const SEVERITY_STYLES: Record<string, string> = {
    32→  Low: "bg-yellow-500/10 text-yellow-600 border-yellow-500/20",
    33→  Medium: "bg-orange-500/10 text-orange-600 border-orange-500/20",
    34→  High: "bg-red-500/10 text-red-600 border-red-500/20",
    35→};
    36→
    37→const STATUS_STYLES: Record<string, string> = {
    38→  Open: "bg-red-500/10 text-red-600",
    39→  Resolved: "bg-emerald-500/10 text-emerald-600",
    40→  Escalated: "bg-purple-500/10 text-purple-600",
    41→};
    42→
    43→const INCIDENT_TYPES = ["Late Arrival", "Absence", "Misbehavior", "Bullying", "Academic", "Uniform Violation", "Other"];
    44→
    45→export function DisciplineModule() {
    46→  const [addOpen, setAddOpen] = useState(false);
    47→  const { moduleData, addModuleItem, updateModuleItem, students } = useAppStore();
    48→  const incidents = (moduleData.incidents || []) as Incident[];
    49→  const [filter, setFilter] = useState<string>("all");
    50→  const [form, setForm] = useState({ studentId: "", studentName: "", class: "", type: "Late Arrival", severity: "Low", description: "", action: "", reportedBy: "" });
    51→
    52→  const filtered = filter === "all" ? incidents : incidents.filter(i => i.status === filter);
    53→  const openCount = incidents.filter(i => i.status === "Open").length;
    54→  const escalatedCount = incidents.filter(i => i.status === "Escalated").length;
    55→  const resolvedCount = incidents.filter(i => i.status === "Resolved").length;
    56→  const highSeverity = incidents.filter(i => i.severity === "High").length;
    57→
    58→  const handleAdd = () => {
    59→    if (!form.studentName || !form.description) return;
    60→    const newIncident: Incident = {
    61→      id: `INC-${String(Date.now()).slice(-6)}`,
    62→      studentId: form.studentId || "—",
    63→      studentName: form.studentName,
    64→      class: form.class || "—",
    65→      date: new Date().toISOString().split("T")[0],
    66→      type: form.type as any,
    67→      severity: form.severity as any,
    68→      description: form.description,
    69→      action: form.action || "Pending review",
    70→      reportedBy: form.reportedBy || "Admin",
    71→      status: "Open",
    72→      parentNotified: false,
    73→    };
    74→    addModuleItem("incidents", newIncident);
    75→    setAddOpen(false);
    76→    setForm({ studentId: "", studentName: "", class: "", type: "Late Arrival", severity: "Low", description: "", action: "", reportedBy: "" });
    77→  };
    78→
    79→  return (
    80→    <div className="space-y-3 fade-in">
    81→      <div className="flex justify-between items-center flex-wrap gap-3">
    82→        <p className="text-sm text-muted-foreground">Track behavioral incidents, warnings, and parent notifications.</p>
    83→        <Button className="bg-primary hover:bg-primary/90" onClick={() => setAddOpen(true)}><Plus className="w-4 h-4 mr-1.5" />Log Incident</Button>
    84→      </div>
    85→
    86→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
    87→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-red-500/10 p-1.5"><AlertTriangle className="w-4 h-4 text-red-600" /></div><div><p className="text-xs text-muted-foreground">Open Cases</p><p className="text-xl font-bold">{openCount}</p></div></CardContent></Card>
    88→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-purple-500/10 p-1.5"><Shield className="w-4 h-4 text-purple-600" /></div><div><p className="text-xs text-muted-foreground">Escalated</p><p className="text-xl font-bold">{escalatedCount}</p></div></CardContent></Card>
    89→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-emerald-500/10 p-1.5"><CheckCircle className="w-4 h-4 text-emerald-600" /></div><div><p className="text-xs text-muted-foreground">Resolved</p><p className="text-xl font-bold">{resolvedCount}</p></div></CardContent></Card>
    90→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-orange-500/10 p-1.5"><AlertTriangle className="w-5 h-5 text-orange-600" /></div><div><p className="text-xs text-muted-foreground">High Severity</p><p className="text-xl font-bold">{highSeverity}</p></div></CardContent></Card>
    91→      </div>
    92→
    93→      <div className="flex items-center gap-2 flex-wrap">
    94→        {["all", "Open", "Escalated", "Resolved"].map(t => (
    95→          <button key={t} onClick={() => setFilter(t)} className={cn("px-3 py-1.5 rounded-lg text-xs font-medium transition-all", filter === t ? "bg-primary text-primary-foreground" : "bg-muted/40 text-muted-foreground hover:bg-muted")}>
    96→            {t === "all" ? "All Incidents" : t}
    97→          </button>
    98→        ))}
    99→      </div>
   100→
   101→      <div className="space-y-3">
   102→        {filtered.map(inc => (
   103→          <Card key={inc.id} className="border-0 shadow-sm glass-card card-hover">
   104→            <CardContent className="p-4">
   105→              <div className="flex items-start gap-4">
   106→                <div className={cn("w-10 h-10 rounded-xl flex items-center justify-center shrink-0",
   107→                  inc.severity === "High" ? "bg-red-500/15" : inc.severity === "Medium" ? "bg-orange-500/15" : "bg-yellow-500/15")}>
   108→                  <AlertTriangle className={cn("w-5 h-5", inc.severity === "High" ? "text-red-600" : inc.severity === "Medium" ? "text-orange-600" : "text-yellow-600")} />
   109→                </div>
   110→                <div className="flex-1 min-w-0">
   111→                  <div className="flex items-center gap-2 flex-wrap">
   112→                    <p className="font-semibold text-sm">{inc.studentName}</p>
   113→                    <span className="text-[10px] text-muted-foreground">{inc.class}</span>
   114→                    <Badge variant="outline" className={cn("text-[9px] border", SEVERITY_STYLES[inc.severity])}>{inc.severity}</Badge>
   115→                    <Badge variant="outline" className="text-[9px]">{inc.type}</Badge>
   116→                    <Badge variant="outline" className={cn("text-[9px] border-0", STATUS_STYLES[inc.status])}>{inc.status}</Badge>
   117→                    {inc.parentNotified && <Badge variant="outline" className="text-[9px] bg-primary/10 text-primary border-0"><Bell className="w-2 h-2 mr-0.5 inline" />Parent Notified</Badge>}
   118→                  </div>
   119→                  <p className="text-sm text-muted-foreground mt-1">{inc.description}</p>
   120→                  <div className="flex items-center gap-4 mt-2 text-xs">
   121→                    <span className="text-muted-foreground flex items-center gap-1"><Calendar className="w-3 h-3" />{inc.date}</span>
   122→                    <span className="text-muted-foreground flex items-center gap-1"><User className="w-3 h-3" />Reported by {inc.reportedBy}</span>
   123→                    <span className="text-primary font-medium flex items-center gap-1"><Zap className="w-3 h-3" />{inc.action}</span>
   124→                  </div>
   125→                  {/* Action buttons */}
   126→                  {inc.status !== "Resolved" && (
   127→                    <div className="flex items-center gap-2 mt-3 pt-3 border-t border-border/30">
   128→                      {inc.status === "Open" && (
   129→                        <Button size="sm" variant="outline" className="h-7 text-xs" onClick={() => updateModuleItem("incidents", inc.id, { status: "Escalated" })}>
   130→                          <AlertTriangle className="w-3 h-3 mr-1" />Escalate
   131→                        </Button>
   132→                      )}
   133→                      <Button size="sm" variant="outline" className="h-7 text-xs text-emerald-600" onClick={() => updateModuleItem("incidents", inc.id, { status: "Resolved" })}>
   134→                        <CheckCircle className="w-3 h-3 mr-1" />Resolve
   135→                      </Button>
   136→                      {!inc.parentNotified && (
   137→                        <Button size="sm" variant="outline" className="h-7 text-xs text-primary ml-auto" onClick={() => updateModuleItem("incidents", inc.id, { parentNotified: true })}>
   138→                          <Bell className="w-3 h-3 mr-1" />Notify Parent
   139→                        </Button>
   140→                      )}
   141→                    </div>
   142→                  )}
   143→                </div>
   144→              </div>
   145→            </CardContent>
   146→          </Card>
   147→        ))}
   148→      </div>
   149→
   150→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   151→        <DialogContent className="max-w-md max-h-[92vh] overflow-y-auto">
   152→          <DialogHeader><DialogTitle className="flex items-center gap-2"><AlertTriangle className="w-4 h-4 text-red-600" />Log Discipline Incident</DialogTitle></DialogHeader>
   153→          <div className="space-y-2 py-1">
   154→            <div className="grid grid-cols-2 gap-3">
   155→              <div><Label>Student Name *</Label><Input value={form.studentName} onChange={e => setForm({...form, studentName: e.target.value})} /></div>
   156→              <div><Label>Class</Label><Input value={form.class} onChange={e => setForm({...form, class: e.target.value})} placeholder="Grade 9-A" /></div>
   157→            </div>
   158→            <div className="grid grid-cols-2 gap-3">
   159→              <div><Label>Type</Label><Select value={form.type} onValueChange={v => setForm({...form, type: v})}><SelectTrigger><SelectValue /></SelectTrigger><SelectContent>{INCIDENT_TYPES.map(t => <SelectItem key={t} value={t}>{t}</SelectItem>)}</SelectContent></Select></div>
   160→              <div><Label>Severity</Label><Select value={form.severity} onValueChange={v => setForm({...form, severity: v})}><SelectTrigger><SelectValue /></SelectTrigger><SelectContent><SelectItem value="Low">Low</SelectItem><SelectItem value="Medium">Medium</SelectItem><SelectItem value="High">High</SelectItem></SelectContent></Select></div>
   161→            </div>
   162→            <div><Label>Description *</Label><Input value={form.description} onChange={e => setForm({...form, description: e.target.value})} placeholder="What happened?" /></div>
   163→            <div><Label>Action Taken</Label><Input value={form.action} onChange={e => setForm({...form, action: e.target.value})} placeholder="e.g., Verbal warning, Detention" /></div>
   164→            <div><Label>Reported By</Label><Input value={form.reportedBy} onChange={e => setForm({...form, reportedBy: e.target.value})} placeholder="Teacher name" /></div>
   165→          </div>
   166→          <DialogFooter><Button variant="outline" onClick={() => setAddOpen(false)}>Cancel</Button><Button className="bg-primary hover:bg-primary/90" onClick={handleAdd} disabled={!form.studentName || !form.description}>Log Incident</Button></DialogFooter>
   167→        </DialogContent>
   168→      </Dialog>
   169→    </div>
   170→  );
   171→}

   182→
   183→export function AlumniModule() {
   184→  const [addOpen, setAddOpen] = useState(false);
   185→  const { moduleData, addModuleItem } = useAppStore();
   186→  const alumni = (moduleData.alumni || []) as Alumnus[];
   187→  const [search, setSearch] = useState("");
   188→  const [yearFilter, setYearFilter] = useState("all");
   189→  const [form, setForm] = useState({ name: "", graduationYear: new Date().getFullYear().toString(), lastClass: "Grade 10", currentStatus: "Studying" as Alumnus["currentStatus"], institution: "", profession: "", email: "", phone: "", donationAmount: 0 });
   190→  const { toast } = useToast();
   191→
   192→  const years = Array.from(new Set(alumni.map(a => a.graduationYear))).sort((a, b) => b.localeCompare(a));
   193→  const filtered = alumni
   194→    .filter(a => yearFilter === "all" || a.graduationYear === yearFilter)
   195→    .filter(a => !search || `${a.name} ${a.profession} ${a.institution}`.toLowerCase().includes(search.toLowerCase()));
   196→
   197→  const totalDonations = alumni.reduce((s, a) => s + a.donationAmount, 0);
   198→  const employedCount = alumni.filter(a => a.currentStatus === "Employed").length;
   199→  const studyingCount = alumni.filter(a => a.currentStatus === "Studying").length;
   200→
   201→  const handleAdd = () => {
   202→    if (!form.name.trim()) {
   203→      toast({ title: "Name required", description: "Please enter the alumnus name.", variant: "destructive" });
   204→      return;
   205→    }
   206→    const newAlumnus: Alumnus = {
   207→      id: `ALU-${String(Date.now()).slice(-6)}`,
   208→      name: form.name,
   209→      graduationYear: form.graduationYear,
   210→      lastClass: form.lastClass,
   211→      currentStatus: form.currentStatus,
   212→      institution: form.institution || "—",
   213→      profession: form.profession || "—",
   214→      email: form.email || "—",
   215→      phone: form.phone || "—",
   216→      donationAmount: Number(form.donationAmount) || 0,
   217→    };
   218→    addModuleItem("alumni", newAlumnus);
   219→    toast({ title: "Alumnus Added ✓", description: `${newAlumnus.name} (Class of ${newAlumnus.graduationYear}) added.` });
   220→    setAddOpen(false);
   221→    setForm({ name: "", graduationYear: new Date().getFullYear().toString(), lastClass: "Grade 10", currentStatus: "Studying", institution: "", profession: "", email: "", phone: "", donationAmount: 0 });
   222→  };
   223→
   224→  return (
   225→    <div className="space-y-3 fade-in">
   226→      <div className="flex justify-between items-center flex-wrap gap-3">
   227→        <p className="text-sm text-muted-foreground">Track graduated students, their careers, and donations.</p>
   228→        <Button className="bg-primary hover:bg-primary/90" onClick={() => setAddOpen(true)}><Plus className="w-4 h-4 mr-1.5" />Add Alumnus</Button>
   229→      </div>
   230→
   231→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   232→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-primary/10 p-1.5"><Users className="w-4 h-4 text-primary" /></div><div><p className="text-xs text-muted-foreground">Total Alumni</p><p className="text-xl font-bold">{alumni.length}</p></div></CardContent></Card>
   233→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-emerald-500/10 p-1.5"><Briefcase className="w-4 h-4 text-emerald-600" /></div><div><p className="text-xs text-muted-foreground">Employed</p><p className="text-xl font-bold">{employedCount}</p></div></CardContent></Card>
   234→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-purple-500/10 p-1.5"><GraduationCap className="w-4 h-4 text-purple-600" /></div><div><p className="text-xs text-muted-foreground">Studying</p><p className="text-xl font-bold">{studyingCount}</p></div></CardContent></Card>
   235→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-accent/10 p-1.5"><Award className="w-4 h-4 text-accent" /></div><div><p className="text-xs text-muted-foreground">Donations</p><p className="text-lg font-bold">Rs. {totalDonations.toLocaleString()}</p></div></CardContent></Card>

   238→      <div className="flex items-center gap-3 flex-wrap">

   243→        <Select value={yearFilter} onValueChange={setYearFilter}>
   244→          <SelectTrigger className="w-32 bg-background/60"><SelectValue /></SelectTrigger>

   309→// PARENT PORTAL MODULE (Read-only view)

   311→export function ParentModule() {
   312→  const [searchId, setSearchId] = useState("");
   313→  const [foundStudent, setFoundStudent] = useState<any>(null);
   314→  const [searched, setSearched] = useState(false);
   315→
   316→  const handleSearch = (students: any[]) => {
   317→    const s = students.find(st => st.id.toLowerCase() === searchId.toLowerCase() || st.name.toLowerCase().includes(searchId.toLowerCase()));
   318→    setFoundStudent(s || null);
   319→    setSearched(true);
   320→  };
