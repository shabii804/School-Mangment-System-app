     1→"use client";
     2→
     3→import { useState, useMemo, useEffect } from "react";
     4→import { useAppStore } from "@/lib/sms-store";
     5→import { Card, CardContent } from "@/components/ui/card";
     6→import { Button } from "@/components/ui/button";
     7→import { Input } from "@/components/ui/input";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Avatar, AvatarFallback } from "@/components/ui/avatar";
    10→import {
    11→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    12→} from "@/components/ui/table";
    13→import {
    14→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    15→} from "@/components/ui/dialog";
    16→import {
    17→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    18→} from "@/components/ui/select";
    19→import {
    20→  Search, DollarSign, Receipt, Printer, AlertTriangle, Filter,
    21→  CheckCircle2, Clock, TrendingUp, Wallet, FileDown, Home, BookOpen,
    22→  GraduationCap, IdCard,
    23→} from "lucide-react";
    24→import { cn } from "@/lib/utils";
    25→import { useToast } from "@/hooks/use-toast";
    26→import { printFeeReceipt, printDefaulterList, printFeeCollectionReport } from "@/lib/pdf-service";
    27→
    28→// Hostel fee constant
    29→const HOSTEL_FEE = 3500;
    30→
    31→export function FeesModule() {
    32→  const { students, feeCollections, collectFee, searchQuery, selectedStudentId, school, feeStructure } = useAppStore();
    33→  const { toast } = useToast();
    34→  const HOSTEL_FEE = feeStructure.hostelFee;
    35→  const ADMISSION_MULTIPLIER = feeStructure.admissionMultiplier;
    36→  const [search, setSearch] = useState("");
    37→  const [feeClassFilter, setFeeClassFilter] = useState("all");
    38→  const [selected, setSelected] = useState<typeof students[0] | null>(null);
    39→
    40→  // Breakdown state — combine tuition + hostel in one receipt
    41→  const [includeTuition, setIncludeTuition] = useState(true);
    42→  const [includeHostel, setIncludeHostel] = useState(false);
    43→  const [includeAdmission, setIncludeAdmission] = useState(false);
    44→  const [customAmount, setCustomAmount] = useState<string>(""); // manual entry
    45→  const [useCustom, setUseCustom] = useState(false);
    46→  const [mode, setMode] = useState<"Cash" | "Cheque" | "Bank Transfer">("Cash");
    47→
    48→  const filtered = useMemo(() => {
    49→    const q = (search || searchQuery).toLowerCase();
    50→    return students.filter(s =>
    51→      s.status === "Active" &&
    52→      (feeClassFilter === "all" || s.class === feeClassFilter) &&
    53→      (!q || `${s.name} ${s.id} ${s.fatherName} ${s.fatherPhone}`.toLowerCase().includes(q))
    54→    );
    55→  }, [students, search, searchQuery, feeClassFilter]);
    56→
    57→  // Auto-select student from cross-module navigation (defaulters → collect)
    58→  useEffect(() => {
    59→    if (selectedStudentId) {
    60→      const s = students.find(st => st.id === selectedStudentId);
    61→      if (s) {
    62→        setSelected(s);
    63→        setIncludeTuition(true);
    64→        setIncludeHostel(s.isHosteller);
    65→        setIncludeAdmission(false);
    66→      }
    67→    }
    68→  }, [selectedStudentId, students]);
    69→
    70→  const openCollect = (s: typeof students[0]) => {
    71→    setSelected(s);
    72→    setIncludeTuition(true);
    73→    setIncludeHostel(s.isHosteller);
    74→    setIncludeAdmission(false);
    75→    setUseCustom(false);
    76→    setCustomAmount("");
    77→  };
    78→
    79→  const tuitionAmt = includeTuition ? selected?.monthlyFee || 0 : 0;
    80→  const hostelAmt = includeHostel && selected?.isHosteller ? HOSTEL_FEE : 0;
    81→  const admissionAmt = includeAdmission ? (selected?.monthlyFee || 0) * ADMISSION_MULTIPLIER : 0;
    82→  const checkboxTotal = tuitionAmt + hostelAmt + admissionAmt;
    83→  const customAmt = useCustom && customAmount ? parseInt(customAmount) || 0 : 0;
    84→  const total = useCustom ? customAmt : checkboxTotal;
    85→
    86→  const feeTypeLabel = useCustom
    87→    ? "Manual Entry"
    88→    : [
    89→        includeTuition && "Tuition",
    90→        includeHostel && "Hostel",
    91→        includeAdmission && "Admission",
    92→      ].filter(Boolean).join(" + ") || "Tuition";
    93→
    94→  const submit = async () => {
    95→    if (!selected) return;
    96→    if (total === 0) {
    97→      toast({ title: "Select a fee type", description: "Please check at least one fee to collect.", variant: "destructive" });
    98→      return;
    99→    }
   100→    collectFee(
   101→      selected.id,
   102→      total,
   103→      mode,
   104→      feeTypeLabel,
   105→      useCustom
   106→        ? undefined // no breakdown for manual entry
   107→        : { tuition: tuitionAmt || undefined, hostel: hostelAmt || undefined, admission: admissionAmt || undefined }
   108→    );
   109→
   110→    // Find the just-created collection (most recent)
   111→    const newCollection = {
   112→      id: `FC-${String(feeCollections.length + 1).padStart(4, "0")}`,
   113→      receiptNo: `RCPT-${new Date().toISOString().slice(0, 10).replace(/-/g, "")}-${String(feeCollections.length + 1).padStart(3, "0")}`,
   114→      studentId: selected.id,
   115→      studentName: selected.name,
   116→      class: selected.class,
   117→      date: new Date().toISOString().split("T")[0],
   118→      amount: total,
   119→      mode,
   120→      tuition: useCustom ? undefined : (tuitionAmt || undefined),
   121→      hostel: useCustom ? undefined : (hostelAmt || undefined),
   122→      admission: useCustom ? undefined : (admissionAmt || undefined),
   123→      feeType: feeTypeLabel,
   124→    };
   125→
   126→    toast({
   127→      title: "Fee Collected ✓",
   128→      description: `Rs. ${total.toLocaleString()} from ${selected.name}. Opening print preview...`,
   129→    });
   130→
   131→    // Open real PDF receipt in new tab
   132→    try {
   133→      await printFeeReceipt(newCollection, selected, school);
   134→    } catch (e) {
   135→      toast({ title: "Print failed", description: "Could not generate PDF.", variant: "destructive" });
   136→    }
   137→
   138→    setSelected(null);
   139→  };
   140→
   141→  const reprintReceipt = async (f: typeof feeCollections[0]) => {
   142→    const student = students.find(s => s.id === f.studentId);
   143→    toast({ title: "Opening receipt...", description: f.receiptNo });
   144→    try {
   145→      await printFeeReceipt(f, student, school);
   146→    } catch {
   147→      toast({ title: "Print failed", description: "Could not generate PDF.", variant: "destructive" });
   148→    }
   149→  };
   150→
   151→  const todayCollections = feeCollections.filter(f => f.date === new Date().toISOString().split("T")[0]);
   152→
   153→  return (
   154→    <div className="space-y-3 fade-in">
   155→      {/* KPI row */}
   156→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   157→        <Card className="border-0 shadow-sm glass-card card-hover"><CardContent className="p-3 flex items-center gap-2">
   158→          <div className="rounded-lg bg-emerald-500/10 p-1.5"><DollarSign className="w-4 h-4 text-emerald-600" /></div>
   159→          <div>
   160→            <p className="text-xs text-muted-foreground">Collected Today</p>
   161→            <p className="text-lg font-bold">Rs. {todayCollections.reduce((s, f) => s + f.amount, 0).toLocaleString()}</p>
   162→          </div>
   163→        </CardContent></Card>
   164→        <Card className="border-0 shadow-sm glass-card card-hover"><CardContent className="p-3 flex items-center gap-2">
   165→          <div className="rounded-lg bg-primary/10 p-1.5"><Receipt className="w-4 h-4 text-primary" /></div>
   166→          <div>
   167→            <p className="text-xs text-muted-foreground">Receipts Today</p>
   168→            <p className="text-xl font-bold">{todayCollections.length}</p>
   169→          </div>
   170→        </CardContent></Card>
   171→        <Card className="border-0 shadow-sm glass-card card-hover"><CardContent className="p-3 flex items-center gap-2">
   172→          <div className="rounded-lg bg-accent/10 p-1.5"><Wallet className="w-4 h-4 text-accent" /></div>
   173→          <div>
   174→            <p className="text-xs text-muted-foreground">Total Outstanding</p>
   175→            <p className="text-lg font-bold">Rs. {students.reduce((s, st) => s + st.outstandingAmount, 0).toLocaleString()}</p>
   176→          </div>
   177→        </CardContent></Card>
   178→        <Card className="border-0 shadow-sm glass-card card-hover"><CardContent className="p-3 flex items-center gap-2">
   179→          <div className="rounded-lg bg-red-500/10 p-1.5"><AlertTriangle className="w-4 h-4 text-red-600" /></div>
   180→          <div>
   181→            <p className="text-xs text-muted-foreground">Defaulters</p>
   182→            <p className="text-xl font-bold">{students.filter(s => s.outstandingAmount > 0).length}</p>
   183→          </div>
   184→        </CardContent></Card>
   185→      </div>
   186→
   187→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   188→        {/* Student lookup */}
   189→        <Card className="lg:col-span-2 border-0 shadow-sm glass-card">
   190→          <div className="p-4 border-b border-border/30 flex items-center gap-2 flex-wrap">
   191→            <div className="relative flex-1 min-w-48">
   192→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   193→              <Input
   194→                placeholder="Search student (name, ID, phone)..."
   195→                className="pl-9 bg-background/60"
   196→                value={search}
   197→                onChange={(e) => setSearch(e.target.value)}
   198→                autoFocus
   199→              />
   200→            </div>
   201→            <Select value={feeClassFilter} onValueChange={setFeeClassFilter}>
   202→              <SelectTrigger className="w-36 bg-background/60 h-9"><Filter className="w-3.5 h-3.5 mr-1.5" /><SelectValue /></SelectTrigger>
   203→              <SelectContent>
   204→                <SelectItem value="all">All Classes</SelectItem>
   205→                {Array.from(new Set(students.filter(s => s.status === "Active").map(s => s.class))).sort().map(c => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   206→              </SelectContent>
   207→            </Select>
   208→          </div>
   209→          <div className="max-h-96 overflow-y-auto">
   210→            {filtered.slice(0, 12).map(s => (
   211→              <div
   212→                key={s.id}
   213→                className="flex items-center gap-3 p-3 border-b border-border/20 hover:bg-muted/30 cursor-pointer transition-colors"
   214→                onClick={() => openCollect(s)}
   215→              >
   216→                <Avatar className="w-9 h-9">
   217→                  <AvatarFallback className={cn("text-xs font-semibold", s.gender === "Male" ? "bg-primary/15 text-primary" : "bg-purple-500/15 text-purple-600")}>
   218→                    {s.name.split(" ").map(w => w[0]).join("").slice(0, 2)}
   219→                  </AvatarFallback>
   220→                </Avatar>
   221→                <div className="flex-1 min-w-0">
   222→                  <p className="text-sm font-medium truncate flex items-center gap-2">
   223→                    {s.name}
   224→                    {s.isHosteller && (
   225→                      <Badge variant="outline" className="text-[9px] bg-accent/10 text-accent border-0 px-1 py-0">
   226→                        <Home className="w-2.5 h-2.5 mr-0.5" />Hostel
   227→                      </Badge>
   228→                    )}
   229→                  </p>
   230→                  <p className="text-xs text-muted-foreground">{s.id} • {s.class}-{s.section} • Roll #{s.rollNo}</p>
   231→                </div>
   232→                <div className="text-right">
   233→                  {s.outstandingAmount > 0 ? (
   234→                    <Badge variant="destructive" className="text-[10px]">Rs. {s.outstandingAmount.toLocaleString()}</Badge>
   235→                  ) : (
   236→                    <Badge variant="outline" className="text-[10px] bg-emerald-500/10 text-emerald-600 border-0">Paid</Badge>
   237→                  )}
   238→                </div>
   239→                <Button size="sm" variant="outline" className="h-7 text-xs">Collect</Button>
   240→              </div>
   241→            ))}
   242→            {filtered.length === 0 && (
   243→              <div className="p-8 text-center text-sm text-muted-foreground">No students found.</div>
   244→            )}
   245→          </div>
   246→        </Card>
   247→
   248→        {/* Recent receipts */}
   249→        <Card className="border-0 shadow-sm glass-card">
   250→          <div className="p-4 border-b border-border/30 flex items-center justify-between">
   251→            <p className="text-sm font-semibold">Recent Receipts</p>
   252→            <Button
   253→              variant="ghost"
   254→              size="sm"
   255→              className="h-7 text-xs"
   256→              onClick={() => {
   257→                toast({ title: "Generating report...", description: "Opening PDF..." });
   258→                printFeeCollectionReport(feeCollections.slice(0, 20), school);
   259→              }}
   260→            >
   261→              <FileDown className="w-3 h-3 mr-1" />Report
   262→            </Button>
   263→          </div>
   264→          <div className="max-h-96 overflow-y-auto">
   265→            {feeCollections.slice(0, 10).map(f => (
   266→              <div key={f.id} className="p-3 border-b border-border/20 hover:bg-muted/30 group">
   267→                <div className="flex items-start justify-between gap-2">
   268→                  <div className="flex-1 min-w-0">
   269→                    <p className="text-xs font-mono text-muted-foreground">{f.receiptNo}</p>
   270→                    <p className="text-sm font-medium truncate mt-0.5">{f.studentName}</p>
   271→                    <p className="text-[10px] text-muted-foreground">{f.class} • {f.feeType} • {f.mode}</p>
   272→                  </div>
   273→                  <div className="text-right">
   274→                    <p className="font-bold text-sm text-emerald-600">Rs. {f.amount.toLocaleString()}</p>
   275→                    <p className="text-[10px] text-muted-foreground">{f.date.slice(5)}</p>
   276→                  </div>
   277→                </div>
   278→                <button
   279→                  onClick={() => reprintReceipt(f)}
   280→                  className="mt-1.5 text-[10px] text-primary opacity-0 group-hover:opacity-100 transition-opacity flex items-center gap-1 hover:underline"
   281→                >
   282→                  <Printer className="w-2.5 h-2.5" />Reprint Receipt
   283→                </button>
   284→              </div>
   285→            ))}
   286→          </div>
   287→        </Card>
   288→      </div>
   289→
   290→      {/* Collect dialog with combined fee breakdown */}
   291→      <Dialog open={!!selected} onOpenChange={(o) => !o && setSelected(null)}>
   292→        <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto p-0 gap-0">
   293→          {selected && (
   294→            <>
   295→              {/* Gradient hero header — student info */}
   296→              <div className="relative overflow-hidden rounded-t-xl bg-gradient-to-br from-emerald-600 via-emerald-700 to-primary/70 px-5 py-4 text-white">
   297→                <div className="absolute inset-0 opacity-10 pattern-grid" />
   298→                <div className="absolute -top-8 -right-8 w-40 h-40 bg-white/10 rounded-full blur-3xl" />
   299→                <div className="relative flex items-center gap-4">
   300→                  <Avatar className="w-14 h-14 ring-4 ring-white/30 shadow-xl shrink-0">
   301→                    {selected.photo ? (
   302→                      // eslint-disable-next-line @next/next/no-img-element
   303→                      <img src={selected.photo} alt={selected.name} className="w-full h-full object-cover" />
   304→                    ) : (
   305→                      <AvatarFallback className="text-base font-bold bg-white/20 text-white border-0">
   306→                        {selected.name.split(" ").map(w => w[0]).join("").slice(0, 2)}
   307→                      </AvatarFallback>
   308→                    )}
   309→                  </Avatar>
   310→                  <div className="flex-1 min-w-0">
   311→                    <div className="flex items-center gap-2 flex-wrap">
   312→                      <h2 className="text-lg font-bold tracking-tight truncate">{selected.name}</h2>
   313→                      {selected.isHosteller && (
   314→                        <span className="inline-flex items-center gap-0.5 text-[9px] px-1.5 py-0.5 rounded-full bg-white/20 font-medium">
   315→                          <Home className="w-2.5 h-2.5" />Hosteller
   316→                        </span>
   317→                      )}
   318→                    </div>
   319→                    <p className="text-xs text-white/75 mt-0.5">
   320→                      <span className="font-mono">{selected.id}</span> • {selected.class}-{selected.section} • Roll #{selected.rollNo}
   321→                    </p>
   322→                  </div>
   323→                  <div className="hidden md:flex flex-col items-end text-right shrink-0">
   324→                    <span className="text-[10px] uppercase tracking-wider text-white/60">Receipt #</span>
   325→                    <span className="text-xs font-semibold mt-0.5 font-mono">RCPT-{new Date().toISOString().slice(0, 10).replace(/-/g, "")}-{String(feeCollections.length + 1).padStart(3, "0")}</span>
   326→                  </div>
   327→                </div>
   328→                {/* Quick stats */}
   329→                <div className="relative grid grid-cols-3 gap-3 mt-4 pt-3 border-t border-white/20">
   330→                  <div>
   331→                    <p className="text-[10px] uppercase tracking-wider text-white/60">Monthly Fee</p>
   332→                    <p className="text-base font-bold mt-0.5">Rs. {selected.monthlyFee.toLocaleString()}</p>
   333→                  </div>
   334→                  <div className="border-l border-white/20 pl-3">
   335→                    <p className="text-[10px] uppercase tracking-wider text-white/60">Outstanding</p>
   336→                    <p className="text-base font-bold mt-0.5 text-red-200">Rs. {selected.outstandingAmount.toLocaleString()}</p>
   337→                  </div>
   338→                  <div className="border-l border-white/20 pl-3">
   339→                    <p className="text-[10px] uppercase tracking-wider text-white/60">Status</p>
   340→                    <p className="text-base font-bold mt-0.5">
   341→                      {selected.feePaidThisMonth ? "Paid ✓" : "Pending"}
   342→                    </p>
   343→                  </div>
   344→                </div>
   345→              </div>
   346→
   347→              {/* Form body */}
   348→              <div className="px-5 py-4 space-y-3">
   349→                {/* ─── Section: Fee Selection ─── */}
   350→                <div className="rounded-xl border border-border/60 bg-card overflow-hidden">
   351→                  <div className="flex items-center gap-2 px-3 py-2 bg-primary/5 border-b border-primary/15">
   352→                    <div className="w-7 h-7 rounded-lg bg-primary/15 flex items-center justify-center">
   353→                      <BookOpen className="w-3.5 h-3.5 text-primary" />
   354→                    </div>
   355→                    <p className="text-xs font-semibold text-primary uppercase tracking-wide">Select Fees to Collect</p>
   356→                    <button
   357→                      type="button"
   358→                      onClick={() => setUseCustom(!useCustom)}
   359→                      className={cn(
   360→                        "ml-auto text-[10px] px-2 py-0.5 rounded-full border transition-all font-medium",
   361→                        useCustom
   362→                          ? "bg-accent/10 text-accent border-accent/30"
   363→                          : "bg-muted text-muted-foreground border-border/40 hover:border-accent/30"
   364→                      )}
   365→                    >
   366→                      {useCustom ? "✓ Manual Entry" : "Manual Entry"}
   367→                    </button>
   368→                  </div>
   369→                  <div className="p-3 space-y-2">
   370→                    {!useCustom && (
   371→                      <div className="space-y-2">
   372→                        {/* Tuition */}
   373→                        <button
   374→                          type="button"
   375→                          onClick={() => setIncludeTuition(!includeTuition)}
   376→                          className={cn(
   377→                            "w-full flex items-center gap-3 p-3 rounded-lg border-2 transition-all",
   378→                            includeTuition ? "border-primary bg-primary/5" : "border-border/40 hover:border-primary/30"
   379→                          )}
   380→                        >
   381→                          <div className={cn("w-5 h-5 rounded-md border-2 flex items-center justify-center transition-all shrink-0",
   382→                            includeTuition ? "bg-primary border-primary" : "border-border")}>
   383→                            {includeTuition && <CheckCircle2 className="w-3.5 h-3.5 text-primary-foreground" />}
   384→                          </div>
   385→                          <div className="flex-1 flex items-center gap-2 min-w-0">
   386→                            <div className={cn("w-8 h-8 rounded-lg flex items-center justify-center shrink-0", includeTuition ? "bg-primary/15" : "bg-muted")}>
   387→                              <BookOpen className={cn("w-3.5 h-3.5", includeTuition ? "text-primary" : "text-muted-foreground")} />
   388→                            </div>
   389→                            <div className="text-left min-w-0">
   390→                              <p className="text-sm font-medium truncate">Tuition Fee</p>
   391→                              <p className="text-[10px] text-muted-foreground">Monthly</p>
   392→                            </div>
   393→                          </div>
   394→                          <p className="font-bold text-sm shrink-0">Rs. {selected.monthlyFee.toLocaleString()}</p>
   395→                        </button>
   396→
   397→                        {/* Hostel — only if hosteller */}
   398→                        {selected.isHosteller && (
   399→                          <button
   400→                            type="button"
   401→                            onClick={() => setIncludeHostel(!includeHostel)}
   402→                            className={cn(
   403→                              "w-full flex items-center gap-3 p-3 rounded-lg border-2 transition-all",
   404→                              includeHostel ? "border-accent bg-accent/5" : "border-border/40 hover:border-accent/30"
   405→                            )}
   406→                          >
   407→                            <div className={cn("w-5 h-5 rounded-md border-2 flex items-center justify-center transition-all shrink-0",
   408→                              includeHostel ? "bg-accent border-accent" : "border-border")}>
   409→                              {includeHostel && <CheckCircle2 className="w-3.5 h-3.5 text-accent-foreground" />}
   410→                            </div>
   411→                            <div className="flex-1 flex items-center gap-2 min-w-0">
   412→                              <div className={cn("w-8 h-8 rounded-lg flex items-center justify-center shrink-0", includeHostel ? "bg-accent/15" : "bg-muted")}>
   413→                                <Home className={cn("w-3.5 h-3.5", includeHostel ? "text-accent" : "text-muted-foreground")} />
   414→                              </div>
   415→                              <div className="text-left min-w-0">
   416→                                <p className="text-sm font-medium truncate">Hostel Fee</p>
   417→                                <p className="text-[10px] text-muted-foreground">Monthly (boarding + meals)</p>
   418→                              </div>
   419→                            </div>
   420→                            <p className="font-bold text-sm shrink-0">Rs. {HOSTEL_FEE.toLocaleString()}</p>
   421→                          </button>
   422→                        )}
   423→
   424→                        {/* Admission */}
   425→                        <button
   426→                          type="button"
   427→                          onClick={() => setIncludeAdmission(!includeAdmission)}
   428→                          className={cn(
   429→                            "w-full flex items-center gap-3 p-3 rounded-lg border-2 transition-all",
   430→                            includeAdmission ? "border-purple-500 bg-purple-500/5" : "border-border/40 hover:border-purple-500/30"
