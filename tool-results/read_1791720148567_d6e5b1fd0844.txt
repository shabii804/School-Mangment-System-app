   271→// HOSTEL MODULE

   273→export function HostelModule() {
   274→  const { moduleData, addModuleItem, updateModuleItem, updateStudent, students } = useAppStore();
   275→  const rooms = (moduleData.hostelRooms || []) as HostelRoom[];
   276→  const allocations = (moduleData.hostelAllocations || []) as any[];
   277→  const [allocOpen, setAllocOpen] = useState(false);
   278→  const [addRoomOpen, setAddRoomOpen] = useState(false);
   279→  const [allocForm, setAllocForm] = useState({ roomId: "", studentId: "", studentName: "" });
   280→  const [roomForm, setRoomForm] = useState({ roomNo: "", capacity: 2, type: "Double" as HostelRoom["type"], floor: 1 });
   281→  const { toast } = useToast();
   282→  const totalCapacity = rooms.reduce((s, r) => s + r.capacity, 0);
   283→  const totalOccupied = rooms.reduce((s, r) => s + r.occupied, 0);
   284→  const availableBeds = totalCapacity - totalOccupied;
   285→
   286→  const handleAddRoom = () => {
   287→    if (!roomForm.roomNo.trim()) {
   288→      toast({ title: "Room number required", description: "Please enter a room number.", variant: "destructive" });
   289→      return;
   290→    }
   291→    addModuleItem("hostelRooms", {
   292→      id: `RM-${String(Date.now()).slice(-6)}`,
   293→      roomNo: roomForm.roomNo,
   294→      capacity: Number(roomForm.capacity) || 1,
   295→      occupied: 0,
   296→      type: roomForm.type,
   297→      floor: Number(roomForm.floor) || 1,
   298→    });
   299→    toast({ title: "Room Added ✓", description: `Room ${roomForm.roomNo} created.` });
   300→    setAddRoomOpen(false);
   301→    setRoomForm({ roomNo: "", capacity: 2, type: "Double", floor: 1 });
   302→  };
   303→
   304→  const handleAllocate = () => {
   305→    if (!allocForm.roomId || !allocForm.studentId.trim()) {
   306→      toast({ title: "Missing info", description: "Please select a room and enter Student ID.", variant: "destructive" });
   307→      return;
   308→    }
