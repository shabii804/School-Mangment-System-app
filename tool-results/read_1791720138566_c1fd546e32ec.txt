   232→const CONCESSION_TYPE_COLORS: Record<string, string> = {
   233→  "Sibling Discount": "bg-blue-500/10 text-blue-600", "Staff Child": "bg-purple-500/10 text-purple-600",

   237→const CONCESSION_STATUS_STYLES: Record<string, string> = {
   238→  Active: "bg-emerald-500/10 text-emerald-600", Expired: "bg-muted text-muted-foreground", Revoked: "bg-red-500/10 text-red-600",
   239→};
   240→
   241→export function ConcessionModule() {
   242→  const { toast } = useToast();
   243→  const { students, moduleData, addModuleItem, updateModuleItem } = useAppStore();
   244→  const concessions = (moduleData.concessions || []) as Concession[];
   245→  const [addOpen, setAddOpen] = useState(false);
   246→  const [form, setForm] = useState({ studentId: "", studentName: "", class: "", type: "Sibling Discount", percentage: 0, fixedAmount: 0, reason: "", validFrom: "", validTo: "" });
   247→  const [selectedClass, setSelectedClass] = useState("all");
   248→  const [selectedStudentId, setSelectedStudentId] = useState("");
   249→
   250→  // Filter students by selected class
   251→  const classOptions = Array.from(new Set(students.filter(s => s.status === "Active").map(s => s.class))).sort();
   252→  const studentOptions = selectedClass === "all"
   253→    ? students.filter(s => s.status === "Active")
   254→    : students.filter(s => s.status === "Active" && s.class === selectedClass);
   255→
   256→  const handleStudentSelect = (id: string) => {
   257→    setSelectedStudentId(id);
   258→    const student = students.find(s => s.id === id);
   259→    if (student) {
   260→      setForm({ ...form, studentId: student.id, studentName: student.name, class: `${student.class}-${student.section}` });
   261→    }
   262→  };
   263→
   264→  const activeCount = concessions.filter(c => c.status === "Active").length;
   265→  const expiredCount = concessions.filter(c => c.status === "Expired").length;
   266→  const totalMonthlySaving = concessions.filter(c => c.status === "Active").reduce((s, c) => s + c.monthlySaving, 0);
   267→  const totalStudents = new Set(concessions.filter(c => c.status === "Active").map(c => c.studentId)).size;
   268→
   269→  const handleAdd = () => {
   270→    if (!form.studentName || !form.reason) return;
   271→    // Look up the student's actual monthly fee for accurate saving calculation
   272→    const student = students.find(s => s.id === form.studentId);
   273→    const monthlyFee = student?.monthlyFee || 0;
   274→    const monthlySaving = form.fixedAmount > 0 ? form.fixedAmount : Math.round((form.percentage / 100) * monthlyFee);
   275→    const newConc: Concession = {
   276→      id: `CON-${String(Date.now()).slice(-6)}`,
   277→      studentId: form.studentId || "—",
   278→      studentName: form.studentName,
   279→      class: form.class || "—",
   280→      type: form.type as any,
   281→      percentage: form.percentage,
   282→      fixedAmount: form.fixedAmount,
   283→      reason: form.reason,
   284→      approvedBy: "Admin",
   285→      status: "Active",
   286→      validFrom: form.validFrom || new Date().toISOString().split("T")[0],
   287→      validTo: form.validTo || "2027-03-31",
   288→      monthlySaving,
   289→    };
   290→    addModuleItem("concessions", newConc);
   291→    toast({ title: "Concession Added ✓", description: `${form.studentName} — ${form.type} (${form.percentage > 0 ? form.percentage + "%" : "Rs. " + form.fixedAmount}) • Saves Rs. ${monthlySaving}/month` });
   292→    setAddOpen(false);
   293→    setForm({ studentId: "", studentName: "", class: "", type: "Sibling Discount", percentage: 0, fixedAmount: 0, reason: "", validFrom: "", validTo: "" });
   294→  };
   295→
   296→  const handleRevoke = (id: string) => {
   297→    if (!window.confirm("Revoke this concession? The student's fee will revert to full amount.")) return;
   298→    updateModuleItem("concessions", id, { status: "Revoked" });
   299→    toast({ title: "Concession Revoked", description: "The concession has been revoked.", variant: "destructive" });
   300→  };
   301→
   302→  return (
   303→    <div className="space-y-3 fade-in">
   304→      <div className="flex justify-between items-center flex-wrap gap-3">
   305→        <p className="text-sm text-muted-foreground">Manage fee discounts, scholarships, and financial aid for students.</p>
   306→        <Button className="bg-primary hover:bg-primary/90" onClick={() => setAddOpen(true)}><Plus className="w-4 h-4 mr-1.5" />Add Concession</Button>
   307→      </div>
   308→
   309→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   310→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-emerald-500/10 p-1.5"><CheckCircle className="w-4 h-4 text-emerald-600" /></div><div><p className="text-xs text-muted-foreground">Active Concessions</p><p className="text-xl font-bold">{activeCount}</p></div></CardContent></Card>
   311→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-primary/10 p-1.5"><Users className="w-4 h-4 text-primary" /></div><div><p className="text-xs text-muted-foreground">Students Benefited</p><p className="text-xl font-bold">{totalStudents}</p></div></CardContent></Card>
   312→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-accent/10 p-1.5"><Wallet className="w-4 h-4 text-accent" /></div><div><p className="text-xs text-muted-foreground">Monthly Saving</p><p className="text-lg font-bold">Rs. {totalMonthlySaving.toLocaleString()}</p></div></CardContent></Card>
   313→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-muted p-2"><Clock className="w-4 h-4 text-muted-foreground" /></div><div><p className="text-xs text-muted-foreground">Expired</p><p className="text-xl font-bold">{expiredCount}</p></div></CardContent></Card>
   314→      </div>
   315→
   316→      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   317→        {concessions.map(c => {
   318→          const Icon = CONCESSION_TYPE_ICONS[c.type] || Gift;
   319→          return (
   320→            <Card key={c.id} className={cn("border-0 shadow-sm glass-card card-hover", c.status === "Revoked" && "opacity-60", c.status === "Expired" && "opacity-70")}>
   321→              <CardContent className="p-4">
   322→                <div className="flex items-start justify-between">
   323→                  <div className="flex items-center gap-3">
   324→                    <div className={cn("w-10 h-10 rounded-xl flex items-center justify-center", CONCESSION_TYPE_COLORS[c.type])}>
   325→                      <Icon className="w-5 h-5" />
   326→                    </div>
   327→                    <div>
   328→                      <p className="font-semibold text-sm">{c.studentName}</p>
   329→                      <p className="text-[10px] text-muted-foreground">{c.class} • {c.studentId}</p>
   330→                    </div>
   331→                  </div>
   332→                  <Badge variant="outline" className={cn("text-[9px] border-0", CONCESSION_STATUS_STYLES[c.status])}>{c.status}</Badge>
