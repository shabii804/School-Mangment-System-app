    36→
    37→const STATUS_STYLES: Record<string, string> = {

    40→  Escalated: "bg-purple-500/10 text-purple-600",

    43→const INCIDENT_TYPES = ["Late Arrival", "Absence", "Misbehavior", "Bullying", "Academic", "Uniform Violation", "Other"];

    55→  const highSeverity = incidents.filter(i => i.severity === "High").length;
    56→
    57→  const handleAdd = () => {
    58→    if (!form.studentName || !form.description) return;
    59→    const newIncident: Incident = {
    60→      id: `INC-${String(Date.now()).slice(-6)}`,
    61→      studentId: form.studentId || "—",
    62→      studentName: form.studentName,
    63→      class: form.class || "—",
    64→      date: new Date().toISOString().split("T")[0],
    65→      type: form.type as any,
    66→      severity: form.severity as any,
    67→      description: form.description,
    68→      action: form.action || "Pending review",
    69→      reportedBy: form.reportedBy || "Admin",
    70→      status: "Open",
    71→      parentNotified: false,
    72→    };
    73→    setIncidents([newIncident, ...incidents]);
    74→    setAddOpen(false);
    75→    setForm({ studentId: "", studentName: "", class: "", type: "Late Arrival", severity: "Low", description: "", action: "", reportedBy: "" });
    76→  };
    77→
    78→  return (
    79→    <div className="space-y-3 fade-in">
    80→      <div className="flex justify-between items-center flex-wrap gap-3">
    81→        <p className="text-sm text-muted-foreground">Track behavioral incidents, warnings, and parent notifications.</p>
    82→        <Button className="bg-primary hover:bg-primary/90" onClick={() => setAddOpen(true)}><Plus className="w-4 h-4 mr-1.5" />Log Incident</Button>

    84→
    85→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

    87→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-purple-500/10 p-1.5"><Shield className="w-4 h-4 text-purple-600" /></div><div><p className="text-xs text-muted-foreground">Escalated</p><p className="text-xl font-bold">{escalatedCount}</p></div></CardContent></Card>
    88→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-emerald-500/10 p-1.5"><CheckCircle className="w-4 h-4 text-emerald-600" /></div><div><p className="text-xs text-muted-foreground">Resolved</p><p className="text-xl font-bold">{resolvedCount}</p></div></CardContent></Card>
    89→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-orange-500/10 p-1.5"><AlertTriangle className="w-5 h-5 text-orange-600" /></div><div><p className="text-xs text-muted-foreground">High Severity</p><p className="text-xl font-bold">{highSeverity}</p></div></CardContent></Card>
    90→      </div>
    91→
    92→      <div className="flex items-center gap-2 flex-wrap">
    93→        {["all", "Open", "Escalated", "Resolved"].map(t => (
    94→          <button key={t} onClick={() => setFilter(t)} className={cn("px-3 py-1.5 rounded-lg text-xs font-medium transition-all", filter === t ? "bg-primary text-primary-foreground" : "bg-muted/40 text-muted-foreground hover:bg-muted")}>
    95→            {t === "all" ? "All Incidents" : t}
    96→          </button>
    97→        ))}
    98→      </div>
    99→
   100→      <div className="space-y-3">
   101→        {filtered.map(inc => (
   102→          <Card key={inc.id} className="border-0 shadow-sm glass-card card-hover">
   103→            <CardContent className="p-4">
   104→              <div className="flex items-start gap-4">
   105→                <div className={cn("w-10 h-10 rounded-xl flex items-center justify-center shrink-0",
   106→                  inc.severity === "High" ? "bg-red-500/15" : inc.severity === "Medium" ? "bg-orange-500/15" : "bg-yellow-500/15")}>
   107→                  <AlertTriangle className={cn("w-5 h-5", inc.severity === "High" ? "text-red-600" : inc.severity === "Medium" ? "text-orange-600" : "text-yellow-600")} />
   108→                </div>
   109→                <div className="flex-1 min-w-0">
   110→                  <div className="flex items-center gap-2 flex-wrap">
   111→                    <p className="font-semibold text-sm">{inc.studentName}</p>
   112→                    <span className="text-[10px] text-muted-foreground">{inc.class}</span>
   113→                    <Badge variant="outline" className={cn("text-[9px] border", SEVERITY_STYLES[inc.severity])}>{inc.severity}</Badge>
   114→                    <Badge variant="outline" className="text-[9px]">{inc.type}</Badge>
   115→                    <Badge variant="outline" className={cn("text-[9px] border-0", STATUS_STYLES[inc.status])}>{inc.status}</Badge>
   116→                    {inc.parentNotified && <Badge variant="outline" className="text-[9px] bg-primary/10 text-primary border-0"><Bell className="w-2 h-2 mr-0.5 inline" />Parent Notified</Badge>}

   123→                  </div>
   124→                </div>
   125→              </div>
   126→            </CardContent>
   127→          </Card>
   128→        ))}

   164→export function AlumniModule() {

   166→  const [alumni, setAlumni] = useState<Alumnus[]>(ALUMNI);
   167→  const [search, setSearch] = useState("");

   170→  const { toast } = useToast();
   171→
   172→  const years = Array.from(new Set(alumni.map(a => a.graduationYear))).sort((a, b) => b.localeCompare(a));
   173→  const filtered = alumni
   174→    .filter(a => yearFilter === "all" || a.graduationYear === yearFilter)
   175→    .filter(a => !search || `${a.name} ${a.profession} ${a.institution}`.toLowerCase().includes(search.toLowerCase()));
   176→
   177→  const totalDonations = alumni.reduce((s, a) => s + a.donationAmount, 0);
   178→  const employedCount = alumni.filter(a => a.currentStatus === "Employed").length;
   179→  const studyingCount = alumni.filter(a => a.currentStatus === "Studying").length;
   180→
   181→  const handleAdd = () => {
   182→    if (!form.name.trim()) {
   183→      toast({ title: "Name required", description: "Please enter the alumnus name.", variant: "destructive" });
   184→      return;
   185→    }
   186→    const newAlumnus: Alumnus = {
   187→      id: `ALU-${String(Date.now()).slice(-6)}`,
   188→      name: form.name,

   190→      lastClass: form.lastClass,
   191→      currentStatus: form.currentStatus,
   192→      institution: form.institution || "—",
   193→      profession: form.profession || "—",
   194→      email: form.email || "—",
   195→      phone: form.phone || "—",
   196→      donationAmount: Number(form.donationAmount) || 0,
   197→    };

   201→    setForm({ name: "", graduationYear: new Date().getFullYear().toString(), lastClass: "Grade 10", currentStatus: "Studying", institution: "", profession: "", email: "", phone: "", donationAmount: 0 });

   207→        <p className="text-sm text-muted-foreground">Track graduated students, their careers, and donations.</p>

   213→        <Card className="border-0 shadow-sm glass-card"><CardContent className="p-3 flex items-center gap-2"><div className="rounded-lg bg-emerald-500/10 p-1.5"><Briefcase className="w-4 h-4 text-emerald-600" /></div><div><p className="text-xs text-muted-foreground">Employed</p><p className="text-xl font-bold">{employedCount}</p></div></CardContent></Card>

   289→// PARENT PORTAL MODULE (Read-only view)

   291→export function ParentModule() {
   292→  const [searchId, setSearchId] = useState("");
   293→  const [foundStudent, setFoundStudent] = useState<any>(null);
   294→  const [searched, setSearched] = useState(false);
   295→
   296→  const handleSearch = (students: any[]) => {
   297→    const s = students.find(st => st.id.toLowerCase() === searchId.toLowerCase() || st.name.toLowerCase().includes(searchId.toLowerCase()));
   298→    setFoundStudent(s || null);
   299→    setSearched(true);
   300→  };
   301→
   302→  return (
   303→    <ParentPortal searchId={searchId} setSearchId={setSearchId} foundStudent={foundStudent} searched={searched} onSearch={handleSearch} />
   304→  );
   305→}
   306→
   307→function ParentPortal({ searchId, setSearchId, foundStudent, searched, onSearch }: any) {
   308→  const { students, feeCollections, school } = useAppStore();
   309→
   310→  return (
   311→    <div className="space-y-3 fade-in">
   312→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-primary via-primary to-primary/80 p-6 text-primary-foreground shadow-lg">
   313→        <div className="absolute inset-0 opacity-10 pattern-grid" />
   314→        <div className="absolute top-0 right-0 w-64 h-64 bg-accent/20 rounded-full blur-3xl -translate-y-1/3 translate-x-1/4" />
   315→        <div className="relative">
   316→          <h2 className="text-2xl font-bold">{school.name} — Parent Portal</h2>
