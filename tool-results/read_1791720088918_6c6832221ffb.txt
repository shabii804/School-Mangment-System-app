     8→// ─── Local date helper ───
     9→// Returns YYYY-MM-DD in LOCAL time (not UTC).
    10→// Using toISOString().split("T")[0] returns UTC date which can be off by 1 day
    11→// in timezones east of GMT (e.g., Pakistan UTC+5 at 4am → previous day in UTC).
    12→export function localDateStr(): string {
    13→  const d = new Date();
    14→  const y = d.getFullYear();
    15→  const m = String(d.getMonth() + 1).padStart(2, "0");
    16→  const day = String(d.getDate()).padStart(2, "0");
    17→  return `${y}-${m}-${day}`;
    18→}
    19→
    20→// Returns YYYY-MM (current local month)
    21→export function localMonthStr(): string {
    22→  return localDateStr().slice(0, 7);
    23→}

   263→function loadEnabledModules(): Record<ModuleKey, boolean> {
   264→  if (typeof window === "undefined") return DEFAULT_ENABLED_MODULES;
   265→  try {
   266→    const raw = window.localStorage.getItem("edudesk_modules");
   267→    if (raw) {
   268→      const parsed = JSON.parse(raw);
   269→      // Merge with defaults to handle new modules added in updates
   270→      return { ...DEFAULT_ENABLED_MODULES, ...parsed };
   271→    }
   272→  } catch {}
   273→  return DEFAULT_ENABLED_MODULES;
   274→}

   279→}

   310→// Generic persistence helpers for collections
   311→function loadCollection<T>(key: string, fallback: T): T {
   312→  if (typeof window === "undefined") return fallback;
   313→  try {
   314→    const raw = window.localStorage.getItem(`edudesk_${key}`);
   315→    if (raw) return JSON.parse(raw) as T;
   316→  } catch {}
   317→  return fallback;

   325→const INITIAL_NOTIFICATIONS: Notification[] = [];
   326→
   327→// === CRITICAL: Clear old seed data BEFORE store creation ===
   328→// This runs once on page load — if version changed, wipe ALL old data
   329→if (typeof window !== "undefined") {
   330→  const CURRENT_VERSION = "12.0-qa-complete";
   331→  const storedVersion = window.localStorage.getItem("edudesk_version");
   332→  if (storedVersion !== CURRENT_VERSION) {
   333→    const keysToRemove: string[] = [];
   334→    for (let i = 0; i < window.localStorage.length; i++) {
   335→      const k = window.localStorage.key(i);
   336→      if (k && k.startsWith("edudesk_") && k !== "edudesk_version") {
   337→        keysToRemove.push(k);
   338→      }
   339→    }
   340→    keysToRemove.forEach(k => { try { window.localStorage.removeItem(k); } catch {} });
   341→    window.localStorage.setItem("edudesk_version", CURRENT_VERSION);
   342→    // Reset the auto-sync guard so the new session doesn't overwrite seed data
   343→    isInitialLoadComplete = false;
   344→  }
   345→
   346→  // ─── Auto-reset feePaidThisMonth on month boundary ───
   347→  // On every page load, check if the current calendar month differs from
   348→  // the students' lastPaidMonth. If so, reset feePaidThisMonth = false
   349→  // and recharge outstandingAmount for the new month.
   350→  // This prevents the "all students show Paid in January" bug.
   351→  const currentMonth = new Date().toISOString().slice(0, 7); // YYYY-MM
   352→  const lastResetMonth = window.localStorage.getItem("edudesk_lastFeeReset");
   353→  if (lastResetMonth !== currentMonth) {
   354→    // Month has changed (or first run) — reset fee flags for all students
   355→    try {
   356→      const rawStudents = window.localStorage.getItem("edudesk_students");
   357→      if (rawStudents) {
   358→        const studentsList = JSON.parse(rawStudents) as Student[];
   359→        let changed = false;
   360→        const updated = studentsList.map(s => {
   361→          if (s.status === "Active" && s.feePaidThisMonth) {
   362→            changed = true;
   363→            return {
   364→              ...s,
   365→              feePaidThisMonth: false,
   366→              // Add this month's fee to outstanding (since it hasn't been paid)
   367→              outstandingAmount: s.outstandingAmount + s.monthlyFee,
   368→              lastPaidMonth: s.lastPaidMonth || lastResetMonth || currentMonth,
   369→            };
   370→          }
   371→          return s;
   372→        });
   373→        if (changed) {
   374→          window.localStorage.setItem("edudesk_students", JSON.stringify(updated));
   375→          console.log(`[Auto-Reset] Fee flags reset for new month: ${currentMonth} (was ${lastResetMonth || "never"})`);
   376→        }
   377→      }
   378→      window.localStorage.setItem("edudesk_lastFeeReset", currentMonth);
   379→    } catch {}
   380→  }

   383→export const useAppStore = create<AppState>((set, get) => ({
   384→  role: "admin",
   385→  setRole: (role) => set({ role, userName: ROLE_NAMES[role] }),
   386→  userName: ROLE_NAMES.admin,
   387→
   388→  activeModule: "dashboard",
   389→  setActiveModule: (m) => set({ activeModule: m }),
   390→
   391→  // Module visibility (on/off)
   392→  enabledModules: loadEnabledModules(),
   393→  toggleModule: (m) => set((state) => {
   394→    if (ALWAYS_ON_MODULES.includes(m)) return state; // skip always-on
   395→    const next = { ...state.enabledModules, [m]: !state.enabledModules[m] };
   396→    saveEnabledModules(next);
   397→    return { enabledModules: next };
   398→  }),
   399→  setModuleEnabled: (m, enabled) => set((state) => {
   400→    if (ALWAYS_ON_MODULES.includes(m)) return state;
   401→    const next = { ...state.enabledModules, [m]: enabled };
   402→    saveEnabledModules(next);
   403→    return { enabledModules: next };
   404→  }),
   405→  setAllModulesEnabled: (enabled) => set((state) => {
   406→    const next = { ...state.enabledModules };
   407→    (Object.keys(next) as ModuleKey[]).forEach(k => {
   408→      if (!ALWAYS_ON_MODULES.includes(k)) next[k] = enabled;
   409→    });
   410→    saveEnabledModules(next);
   411→    return { enabledModules: next };
   412→  }),
   413→
   414→  school: loadSchool(),
   415→  updateSchool: (patch) => {
   416→    const next = { ...get().school, ...patch };
   417→    saveSchool(next);
   418→    set({ school: next });
   419→  },
   420→  setLogo: (logo) => {
   421→    const next = { ...get().school, logo };
   422→    saveSchool(next);
   423→    set({ school: next });
   424→  },
   425→
   426→  feeStructure: loadFeeStructure(),
   427→  updateFeeStructure: (patch) => {
   428→    const next = { ...get().feeStructure, ...patch };
   429→    saveFeeStructure(next);
   430→    set({ feeStructure: next });
   431→  },
   432→  setClassFee: (className, amount) => {
   433→    const next = { ...get().feeStructure, classFees: { ...get().feeStructure.classFees, [className]: amount } };
   434→    saveFeeStructure(next);
   435→    set({ feeStructure: next });
   436→  },
   437→
   438→  classes: loadCollection("classes", []),
   439→  subjects: loadCollection("subjects", []),
   440→  addClass: (cls) => set((state) => ({ classes: [...state.classes, cls] })),
   441→  updateClass: (name, patch) => set((state) => ({ classes: state.classes.map(c => c.name === name ? { ...c, ...patch } : c) })),
   442→  deleteClass: (name) => set((state) => ({ classes: state.classes.filter(c => c.name !== name) })),
   443→  addSubject: (subject) => set((state) => ({ subjects: [...state.subjects, subject] })),
   444→  deleteSubject: (subject) => set((state) => ({ subjects: state.subjects.filter(s => s !== subject) })),
   445→
   446→  students: loadCollection("students", STUDENTS),
   447→  teachers: loadCollection("teachers", TEACHERS),
   448→  staff: loadCollection("staff", STAFF),
   449→  feeCollections: loadCollection("feeCollections", FEE_COLLECTIONS),
   450→  activities: loadCollection("activities", ACTIVITIES),
   451→
   452→  isDark: false,
   453→  toggleTheme: () => set((s) => ({ isDark: !s.isDark })),
   454→
   455→  // ─── Database sync ───
   456→  syncStatus: "idle",
   457→  lastSyncAt: null,
   458→  dbStats: null,
   459→  setSyncStatus: (s) => set({ syncStatus: s }),
   460→  syncToDatabase: async () => {
   461→    set({ syncStatus: "syncing" });
   462→    try {
   463→      const state = get();
   464→      const stateToSave = {
   465→        students: state.students,
   466→        teachers: state.teachers,
   467→        staff: state.staff,
   468→        feeCollections: state.feeCollections,
   469→        activities: state.activities,
   470→        documents: state.documents,
   471→        commLogs: state.commLogs,
   472→        transactions: state.transactions,
   473→        classes: state.classes,
   474→        subjects: state.subjects,
   475→        moduleData: state.moduleData,
   476→        attendanceRecords: state.attendanceRecords,
   477→        school: state.school,
   478→        feeStructure: state.feeStructure,
   479→        enabledModules: state.enabledModules,
   480→      };
   481→      const res = await fetch("/api/state", {
   482→        method: "POST",
   483→        headers: { "Content-Type": "application/json" },
   484→        body: JSON.stringify({ state: stateToSave }),
   485→      });
   486→      const data = await res.json();
   487→      if (data.success) {
   488→        set({ syncStatus: "online", lastSyncAt: new Date().toISOString(), dbStats: data.stats });
   489→        return true;
   490→      } else {
   491→        set({ syncStatus: "error" });
   492→        return false;
   493→      }
   494→    } catch (e) {
   495→      // Network error — we're offline, data is still in localStorage
   496→      set({ syncStatus: "offline" });
   497→      return false;
   498→    }
   499→  },
   500→  loadFromDatabase: async () => {
   501→    // Set the flag IMMEDIATELY (synchronously, before any async work)
   502→    // This prevents the subscribe block from syncing empty state to SQLite
   503→    // before we've had a chance to load the real data
   504→    isInitialLoadStarted = true;
   505→    set({ syncStatus: "syncing" });
   506→    try {
   507→      const res = await fetch("/api/state", { method: "GET" });
   508→      const data = await res.json();
   509→      if (data.success && data.state) {
   510→        const s = data.state;
   511→        // Only update collections that exist in the database (don't overwrite with undefined)
   512→        const patch: Partial<AppState> = { syncStatus: "online", dbStats: data.stats, lastSyncAt: new Date().toISOString() };
   513→        if (s.students) patch.students = s.students;
   514→        if (s.teachers) patch.teachers = s.teachers;
   515→        if (s.staff) patch.staff = s.staff;
   516→        if (s.feeCollections) patch.feeCollections = s.feeCollections;
   517→        if (s.activities) patch.activities = s.activities;
   518→        if (s.documents) patch.documents = s.documents;
   519→        if (s.commLogs) patch.commLogs = s.commLogs;
   520→        if (s.transactions) patch.transactions = s.transactions;
   521→        if (s.classes) patch.classes = s.classes;
   522→        if (s.subjects) patch.subjects = s.subjects;
   523→        if (s.moduleData) patch.moduleData = s.moduleData;
   524→        if (s.attendanceRecords) patch.attendanceRecords = s.attendanceRecords;
   525→        if (s.school) patch.school = s.school;
   526→        if (s.feeStructure) patch.feeStructure = s.feeStructure;
   527→        if (s.enabledModules) patch.enabledModules = s.enabledModules;
   528→        set(patch);
   529→        // Also update localStorage as cache
   530→        if (typeof window !== "undefined") {
   531→          Object.entries(s).forEach(([key, value]) => {
   532→            try { window.localStorage.setItem(`edudesk_${key}`, JSON.stringify(value)); } catch {}
   533→          });
