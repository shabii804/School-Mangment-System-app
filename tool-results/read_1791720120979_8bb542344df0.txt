   423→  const SUBJECTS = storeSubjects;
   424→  const classOptions = CLASSES.length > 0
   425→    ? CLASSES.flatMap(c => (c.sections.length > 0 ? c.sections : ["A"]).map(s => `${c.name}-${s}`))
   426→    : [];
   427→
   428→  const [selectedClass, setSelectedClass] = useState(classOptions[0] || "");
   429→  const [editSlot, setEditSlot] = useState<{ day: string; period: number; slot?: TimetableSlot } | null>(null);
   430→  const [aiOpen, setAiOpen] = useState(false);

   432→
   433→  // Read timetable from store moduleData.timetables (keyed by class-section)
   434→  const timetables = (moduleData.timetables || {}) as Record<string, TimetableSlot[]>;
   435→  const slots: TimetableSlot[] = timetables[selectedClass] || [];
   436→
   437→  const getSlot = (day: string, period: number) => slots.find(s => s.day === day && s.period === period);
   438→
   439→  // Save a slot (add or update) — persists to store
   440→  const saveSlot = (day: string, period: number, data: { subject: string; teacher: string; room: string }) => {
   441→    const existing = slots.filter(s => !(s.day === day && s.period === period));
   442→    const updated = [...existing, { day, period, ...data }];
   443→    setModuleData("timetables", { ...timetables, [selectedClass]: updated });
   444→  };
   445→
   446→  // Delete a slot
   447→  const deleteSlot = (day: string, period: number) => {
   448→    const updated = slots.filter(s => !(s.day === day && s.period === period));
   449→    setModuleData("timetables", { ...timetables, [selectedClass]: updated });
   450→  };
   451→
   452→  // ─── AI Auto-Generate Timetable ───
   453→  // Smart algorithm:
   454→  // 1. Distributes subjects evenly across the week (no subject twice on same day if avoidable)
   455→  // 2. Respects teacher assignments (from teachers' subjects list)
   456→  // 3. Avoids double-booking a teacher in the same period
   457→  // 4. Assigns rooms (defaults to "Room 1" etc., or uses class section)
   458→  const autoGenerate = () => {
   459→    if (!selectedClass) {
   460→      toast({ title: "Select a class", description: "Please select a class first.", variant: "destructive" });
   461→      return;
   462→    }
   463→    if (SUBJECTS.length === 0) {
   464→      toast({ title: "No subjects", description: "Add subjects in the Classes module first.", variant: "destructive" });
   465→      setAiOpen(false);
   466→      return;
   467→    }
   468→    setAiGenerating(true);
   469→
   470→    // Build subject→teacher map from store teachers
   471→    const subjectTeachers: Record<string, string[]> = {};
   472→    SUBJECTS.forEach(subj => {
   473→      const eligible = teachers.filter(t => t.subjects.includes(subj));
   474→      subjectTeachers[subj] = eligible.length > 0 ? eligible.map(t => t.name) : [];
   475→    });
   476→
   477→    // Track teacher bookings: { teacherName: Set of "day-period" }
   478→    const teacherBookings: Record<string, Set<string>> = {};
   479→    teachers.forEach(t => { teacherBookings[t.name] = new Set(); });
   480→
   481→    // Generate schedule
   482→    const newSlots: TimetableSlot[] = [];
   483→    const usedToday: Record<string, Set<string>> = {}; // day → subjects used that day
   484→    DAYS.forEach(d => { usedToday[d] = new Set(); });
   485→
   486→    // Subject quota: each subject should appear roughly (6 periods × 6 days) / numSubjects times
   487→    const totalPeriods = PERIOD_TIMES.length * DAYS.length; // 36
   488→    const subjectQuota = Math.max(1, Math.floor(totalPeriods / SUBJECTS.length));
   489→
   490→    // Count how many times each subject has been scheduled
   491→    const subjectCount: Record<string, number> = {};
   492→    SUBJECTS.forEach(s => { subjectCount[s] = 0; });
   493→
   494→    // Determine the base room from the selected class
   495→    const baseRoom = selectedClass ? `Room ${selectedClass.split("-").pop() || "1"}` : "Room 1";
   496→
   497→    let generated = 0;
   498→    for (const day of DAYS) {
   499→      for (let period = 1; period <= PERIOD_TIMES.length; period++) {
   500→        // Pick a subject: prefer ones that haven't met quota and weren't used today
   501→        const candidates = SUBJECTS
   502→          .filter(s => !usedToday[day].has(s))
   503→          .sort((a, b) => (subjectCount[a] - subjectCount[b]));
   504→
   505→        if (candidates.length === 0) {
   506→          // All subjects used today; pick the least-used overall
   507→          const fallback = SUBJECTS.sort((a, b) => subjectCount[a] - subjectCount[b])[0];
   508→          if (fallback) {
   509→            const teacher = subjectTeachers[fallback]?.[0] || "—";
   510→            newSlots.push({ day, period, subject: fallback, teacher, room: baseRoom });
   511→            if (teacher !== "—") teacherBookings[teacher]?.add(`${day}-${period}`);
   512→            subjectCount[fallback]++;
   513→            generated++;
   514→          }
   515→          continue;
   516→        }
   517→
   518→        const subject = candidates[0];
   519→        // Find a teacher who teaches this subject and isn't booked in this period
   520→        const eligibleTeachers = subjectTeachers[subject] || [];
   521→        let assignedTeacher = "—";
   522→        for (const tName of eligibleTeachers) {
   523→          if (!teacherBookings[tName]?.has(`${day}-${period}`)) {
   524→            assignedTeacher = tName;
   525→            teacherBookings[tName]?.add(`${day}-${period}`);
   526→            break;
   527→          }
   528→        }
   529→        // If all eligible teachers are booked, assign the first one anyway (or "TBD")
   530→        if (assignedTeacher === "—" && eligibleTeachers.length > 0) {
   531→          assignedTeacher = `${eligibleTeachers[0]} (busy)`;
   532→        }
   533→
   534→        newSlots.push({ day, period, subject, teacher: assignedTeacher, room: baseRoom });
   535→        usedToday[day].add(subject);
   536→        subjectCount[subject]++;
   537→        generated++;
   538→      }
   539→    }
   540→
   541→    // Save the generated timetable
   542→    setModuleData("timetables", { ...timetables, [selectedClass]: newSlots });
   543→    setAiGenerating(false);
   544→    setAiOpen(false);
   545→    toast({
   546→      title: "Timetable Auto-Generated ✓",
   547→      description: `${generated} slots created for ${selectedClass} • ${SUBJECTS.length} subjects distributed • ${teachers.length} teachers assigned.`,
   548→    });
   549→  };
   550→
   551→  return (
   552→    <div className="space-y-3">
   553→      <div className="flex justify-between items-center flex-wrap gap-3">
   554→        <div className="flex items-center gap-3">
   555→          <p className="text-sm text-muted-foreground">Weekly class timetable</p>
   556→          <Select value={selectedClass} onValueChange={setSelectedClass}>
   557→            <SelectTrigger className="w-40 bg-background/60"><SelectValue placeholder="Select class..." /></SelectTrigger>
   558→            <SelectContent>
   559→              {classOptions.length === 0 ? (
   560→                <div className="px-3 py-2 text-xs text-muted-foreground text-center">No classes yet. Add in Classes module.</div>
   561→              ) : classOptions.map(c => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   562→            </SelectContent>
   563→          </Select>
   564→        </div>
   565→        <div className="flex gap-2">
   566→          <Button variant="outline" size="sm" onClick={() => printTimetablePDF(selectedClass, slots, school)} disabled={!selectedClass || slots.length === 0}>
   567→            <Printer className="w-4 h-4 mr-1.5" />Print
   568→          </Button>
   569→          <Button variant="outline" size="sm" className="border-accent/30 text-accent hover:bg-accent/10" onClick={() => setAiOpen(true)} disabled={!selectedClass || SUBJECTS.length === 0}>
   570→            <Wand2 className="w-4 h-4 mr-1.5" />AI Auto-Generate
   571→          </Button>
   572→        </div>
   573→      </div>
   574→
   575→      {/* Info bar */}
   576→      <div className="flex flex-wrap items-center gap-3 text-xs">
   577→        <span className="text-muted-foreground">{slots.length} of {DAYS.length * PERIOD_TIMES.length} slots filled</span>
   578→        {SUBJECTS.length === 0 && (

   580→            <AlertCircle className="w-3 h-3 mr-1" />Add subjects first

   590→      {!selectedClass ? (
   591→        <Card className="border-0 shadow-sm glass-card">
   592→          <CardContent className="p-12 text-center">
   593→            <School className="w-10 h-10 mx-auto mb-2 text-muted-foreground/40" />

   757→function SlotEditor({ slot, subjects, teachers, onSave, onDelete, onCancel }: {
   758→  slot?: TimetableSlot;
   759→  subjects: string[];
   760→  teachers: { name: string; subjects: string[] }[];
   761→  onSave: (data: { subject: string; teacher: string; room: string }) => void;

   800→              : teachers.map(t => <SelectItem key={t.name} value={t.name}>{t.name}</SelectItem>)

   803→              <div className="px-3 py-2 text-xs text-muted-foreground text-center">No teacher assigned to {subject} yet.</div>

   812→      </div>

   814→      <DialogFooter className="flex !justify-between gap-2">
   815→        <div>
