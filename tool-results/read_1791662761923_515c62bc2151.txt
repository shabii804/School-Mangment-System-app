   243→   243→function loadEnabledModules(): Record<ModuleKey, boolean> {
   244→   244→  if (typeof window === "undefined") return DEFAULT_ENABLED_MODULES;
   245→   245→  try {
   246→   246→    const raw = window.localStorage.getItem("edudesk_modules");
   247→   247→    if (raw) {
   248→   248→      const parsed = JSON.parse(raw);
   249→   249→      // Merge with defaults to handle new modules added in updates
   250→   250→      return { ...DEFAULT_ENABLED_MODULES, ...parsed };
   251→   251→    }
   252→   252→  } catch {}
   253→   253→  return DEFAULT_ENABLED_MODULES;
   254→   254→}

   325→   325→export const useAppStore = create<AppState>((set, get) => ({

   348→   348→    const next = { ...state.enabledModules };
   349→   349→    (Object.keys(next) as ModuleKey[]).forEach(k => {
   350→   350→      if (!ALWAYS_ON_MODULES.includes(k)) next[k] = enabled;
   351→   351→    });

   353→   353→    return { enabledModules: next };

   384→   384→  deleteClass: (name) => set((state) => ({ classes: state.classes.filter(c => c.name !== name) })),

   402→   402→  syncToDatabase: async () => {
   403→   403→    set({ syncStatus: "syncing" });
   404→   404→    try {
   405→   405→      const state = get();
   406→   406→      const stateToSave = {
   407→   407→        students: state.students,
   408→   408→        teachers: state.teachers,
   409→   409→        staff: state.staff,
   410→   410→        feeCollections: state.feeCollections,
   411→   411→        activities: state.activities,
   412→   412→        documents: state.documents,
   413→   413→        commLogs: state.commLogs,

   418→   418→        attendanceRecords: state.attendanceRecords,
   419→   419→        school: state.school,
   420→   420→        feeStructure: state.feeStructure,
   421→   421→        enabledModules: state.enabledModules,
   422→   422→      };
   423→   423→      const res = await fetch("/api/state", {
   424→   424→        method: "POST",
   425→   425→        headers: { "Content-Type": "application/json" },
   426→   426→        body: JSON.stringify({ state: stateToSave }),
   427→   427→      });
   428→   428→      const data = await res.json();
   429→   429→      if (data.success) {
   430→   430→        set({ syncStatus: "online", lastSyncAt: new Date().toISOString(), dbStats: data.stats });
   431→   431→        return true;
   432→   432→      } else {
   433→   433→        set({ syncStatus: "error" });
   434→   434→        return false;
   435→   435→      }
   436→   436→    } catch (e) {
   437→   437→      // Network error — we're offline, data is still in localStorage
   438→   438→      set({ syncStatus: "offline" });
   439→   439→      return false;
   440→   440→    }
   441→   441→  },
   442→   442→  loadFromDatabase: async () => {
   443→   443→    // Set the flag IMMEDIATELY (synchronously, before any async work)
   444→   444→    // This prevents the subscribe block from syncing empty state to SQLite
   445→   445→    // before we've had a chance to load the real data
   446→   446→    isInitialLoadStarted = true;
   447→   447→    set({ syncStatus: "syncing" });
   448→   448→    try {
   449→   449→      const res = await fetch("/api/state", { method: "GET" });
   450→   450→      const data = await res.json();
   451→   451→      if (data.success && data.state) {
   452→   452→        const s = data.state;
   453→   453→        // Only update collections that exist in the database (don't overwrite with undefined)
   454→   454→        const patch: Partial<AppState> = { syncStatus: "online", dbStats: data.stats, lastSyncAt: new Date().toISOString() };
