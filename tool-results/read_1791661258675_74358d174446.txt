   243→function loadEnabledModules(): Record<ModuleKey, boolean> {
   244→  if (typeof window === "undefined") return DEFAULT_ENABLED_MODULES;
   245→  try {
   246→    const raw = window.localStorage.getItem("edudesk_modules");
   247→    if (raw) {
   248→      const parsed = JSON.parse(raw);
   249→      // Merge with defaults to handle new modules added in updates
   250→      return { ...DEFAULT_ENABLED_MODULES, ...parsed };
   251→    }
   252→  } catch {}
   253→  return DEFAULT_ENABLED_MODULES;
   254→}

   256→function saveEnabledModules(modules: Record<ModuleKey, boolean>) {
   257→  if (typeof window === "undefined") return;
   258→  try { window.localStorage.setItem("edudesk_modules", JSON.stringify(modules)); } catch {}
   259→}

   290→// Generic persistence helpers for collections
   291→function loadCollection<T>(key: string, fallback: T): T {
   292→  if (typeof window === "undefined") return fallback;
   293→  try {
   294→    const raw = window.localStorage.getItem(`edudesk_${key}`);
   295→    if (raw) return JSON.parse(raw) as T;
   296→  } catch {}
   297→  return fallback;
   298→}

   300→function saveCollection<T>(key: string, data: T) {

   303→}

   307→// === CRITICAL: Clear old seed data BEFORE store creation ===
   308→// This runs once on page load — if version changed, wipe ALL old data
   309→if (typeof window !== "undefined") {
   310→  const CURRENT_VERSION = "9.0-seeded";
   311→  const storedVersion = window.localStorage.getItem("edudesk_version");
   312→  if (storedVersion !== CURRENT_VERSION) {
   313→    const keysToRemove: string[] = [];
   314→    for (let i = 0; i < window.localStorage.length; i++) {
   315→      const k = window.localStorage.key(i);
   316→      if (k && k.startsWith("edudesk_") && k !== "edudesk_version") {
   317→        keysToRemove.push(k);
   318→      }
   319→    }
   320→    keysToRemove.forEach(k => { try { window.localStorage.removeItem(k); } catch {} });
   321→    window.localStorage.setItem("edudesk_version", CURRENT_VERSION);
   322→  }

   430→        set({ syncStatus: "online", lastSyncAt: new Date().toISOString(), dbStats: data.stats });

   462→        if (s.transactions) patch.transactions = s.transactions;
   463→        if (s.classes) patch.classes = s.classes;
   464→        if (s.subjects) patch.subjects = s.subjects;
   465→        if (s.moduleData) patch.moduleData = s.moduleData;
   466→        if (s.attendanceRecords) patch.attendanceRecords = s.attendanceRecords;
   467→        if (s.school) patch.school = s.school;
   468→        if (s.feeStructure) patch.feeStructure = s.feeStructure;
   469→        if (s.enabledModules) patch.enabledModules = s.enabledModules;
   470→        set(patch);
   471→        // Also update localStorage as cache
   472→        if (typeof window !== "undefined") {
   473→          Object.entries(s).forEach(([key, value]) => {
   474→            try { window.localStorage.setItem(`edudesk_${key}`, JSON.stringify(value)); } catch {}
   475→          });
   476→        }

   478→        // Use setTimeout to ensure the set() above (which triggers subscribe) completes first
   479→        // with isInitialLoadComplete still false, THEN enable auto-sync
   480→        setTimeout(() => { isInitialLoadComplete = true; }, 500);
   481→        return true;
   482→      }
   483→      // Even if SQLite was empty, enable auto-sync so future changes are saved
   484→      setTimeout(() => { isInitialLoadComplete = true; }, 500);
   485→      set({ syncStatus: "offline" });
   486→      return false;
   487→    } catch (e) {
   488→      // On error, enable auto-sync so localStorage data gets saved to SQLite
   489→      setTimeout(() => { isInitialLoadComplete = true; }, 500);
   490→      set({ syncStatus: "offline" });
   491→      return false;
   492→    }

   651→// Auto-persist key collections to localStorage on every change (instant)
   652→// + debounce-sync to SQLite (source of truth)
   653→// IMPORTANT: Skip SQLite auto-sync until loadFromDatabase() has run, to avoid
   654→// overwriting seeded data with the initial empty state
   655→let sqliteSyncTimer: ReturnType<typeof setTimeout> | null = null;
   656→let isInitialLoadStarted = false; // set true synchronously when loadFromDatabase() is called
   657→let isInitialLoadComplete = false; // set true after loadFromDatabase() finishes
   658→
   659→useAppStore.subscribe((state) => {
   660→  // 1. Instant localStorage save (offline cache) — always do this
   661→  saveCollection("students", state.students);
   662→  saveCollection("teachers", state.teachers);
   663→  saveCollection("staff", state.staff);
   664→  saveCollection("feeCollections", state.feeCollections);

   671→  saveCollection("moduleData", state.moduleData);
   672→  saveCollection("attendanceRecords", state.attendanceRecords);
   673→
   674→  // 2. Debounced SQLite sync (source of truth) — 2 second delay after last change
   675→  // SKIP until initial load from SQLite has completed — prevents overwriting seeded data
   676→  if (!isInitialLoadComplete) return;
   677→  if (sqliteSyncTimer) clearTimeout(sqliteSyncTimer);
   678→  sqliteSyncTimer = setTimeout(async () => {
   679→    try {
   680→      const stateToSave = {

   697→      const res = await fetch("/api/state", {
   698→        method: "POST",
   699→        headers: { "Content-Type": "application/json" },
   700→        body: JSON.stringify({ state: stateToSave }),
   701→      });
   702→      const data = await res.json();
   703→      if (data.success) {

   705→      }
